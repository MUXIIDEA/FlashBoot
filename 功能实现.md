### 4.1 数据模型

#### UsbDevice.h

```cpp
#pragma once
#include "pch.h"

enum class PartitionScheme {
    MBR,
    GPT,
    Unknown
};

enum class FileSystem {
    FAT32,
    NTFS,
    exFAT,
    UDF,
    Unknown
};

enum class DeviceStatus {
    Ready,
    Busy,
    Error,
    WriteProtected,
    Unknown
};

class UsbDevice {
public:
    QString deviceId;           // 设备实例 ID
    QString physicalPath;       // 物理设备路径 (\\\\.\\PhysicalDriveN)
    QString driveLetter;        // 盘符 (E:)
    QString vendor;             // 厂商
    QString product;            // 产品型号
    QString serialNumber;       // 序列号
    quint64 totalBytes = 0;     // 总容量
    quint64 freeBytes = 0;      // 可用空间
    FileSystem fileSystem = FileSystem::Unknown;
    PartitionScheme partitionScheme = PartitionScheme::Unknown;
    DeviceStatus status = DeviceStatus::Unknown;
    bool isRemovable = true;    // 是否可移动设备
    bool isUsb = true;          // 是否 USB 设备
    int diskNumber = -1;        // 物理磁盘号
    
    // 计算已用百分比
    int usedPercent() const {
        if (totalBytes == 0) return 0;
        return static_cast<int>((totalBytes - freeBytes) * 100 / totalBytes);
    }
    
    // 格式化容量显示
    QString capacityString() const {
        return QString("%1 / %2")
            .arg(formatBytes(totalBytes - freeBytes))
            .arg(formatBytes(totalBytes));
    }
    
    static QString formatBytes(quint64 bytes) {
        const char* units[] = {"B", "KB", "MB", "GB", "TB"};
        int unitIndex = 0;
        double size = static_cast<double>(bytes);
        while (size >= 1024.0 && unitIndex < 4) {
            size /= 1024.0;
            unitIndex++;
        }
        return QString("%1 %2").arg(size, 0, 'f', 2).arg(units[unitIndex]);
    }
};

Q_DECLARE_METATYPE(UsbDevice)
```

#### IsoInfo.h

```cpp
#pragma once
#include "pch.h"

enum class BootMode {
    BIOS,
    UEFI,
    Dual,       // BIOS + UEFI
    Unknown
};

enum class ArchType {
    x86,
    x64,
    ARM64,
    Unknown
};

class IsoInfo {
public:
    QString filePath;           // 文件路径
    QString fileName;           // 文件名
    QString displayName;        // 显示名称 (从 ISO 提取)
    QString version;            // 版本号
    QString buildNumber;        // Build 号
    ArchType arch = ArchType::Unknown;
    BootMode bootMode = BootMode::Unknown;
    quint64 fileSize = 0;       // 文件大小
    QByteArray sha256;          // SHA256 哈希
    bool isValid = false;       // 是否有效 ISO
    QString errorString;        // 错误信息
    
    // 从 ISO 文件解析信息 (静态工厂方法)
    static IsoInfo fromFile(const QString& path);
    
    QString archString() const {
        switch (arch) {
            case ArchType::x86: return "x86";
            case ArchType::x64: return "x64";
            case ArchType::ARM64: return "ARM64";
            default: return "Unknown";
        }
    }
    
    QString bootModeString() const {
        switch (bootMode) {
            case BootMode::BIOS: return "Legacy BIOS";
            case BootMode::UEFI: return "UEFI Only";
            case BootMode::Dual: return "BIOS + UEFI";
            default: return "Unknown";
        }
    }
};

Q_DECLARE_METATYPE(IsoInfo)
```

```cpp
// 获取物理磁盘号
int UsbDeviceManager::getPhysicalDiskNumber(char driveLetter) {
    QString volumePath = QString("\\\\.\\%1:").arg(driveLetter);
    HANDLE hVolume = CreateFileW(
        volumePath.toStdWString().c_str(),
        0, FILE_SHARE_READ | FILE_SHARE_WRITE,
        nullptr, OPEN_EXISTING, 0, nullptr
    );
    
    if (hVolume == INVALID_HANDLE_VALUE) return -1;
    
    VOLUME_DISK_EXTENTS extents;
    DWORD bytesReturned;
    int diskNumber = -1;
    
    if (DeviceIoControl(hVolume, IOCTL_VOLUME_GET_VOLUME_DISK_EXTENTS,
                        nullptr, 0,
                        &extents, sizeof(extents),
                        &bytesReturned, nullptr)) {
        if (extents.NumberOfDiskExtents > 0) {
            diskNumber = extents.Extents[0].DiskNumber;
        }
    }
    
    CloseHandle(hVolume);
    return diskNumber;
}

    GUID guid = GUID_DEVINTERFACE_DISK;

    while (SetupDiEnumDeviceInterfaces(hDevInfo, nullptr, &guid, index++, &interfaceData)) {
        DWORD requiredSize = 0;
        SetupDiGetDeviceInterfaceDetailW(hDevInfo, &interfaceData, nullptr, 0, &requiredSize, nullptr);
        
        if (requiredSize == 0) continue;
        
        std::vector<BYTE> detailDataBuffer(requiredSize);
        PSP_DEVICE_INTERFACE_DETAIL_DATA_W detailData = 
            reinterpret_cast<PSP_DEVICE_INTERFACE_DETAIL_DATA_W>(detailDataBuffer.data());
        detailData->cbSize = sizeof(SP_DEVICE_INTERFACE_DETAIL_DATA_W);
        
        SP_DEVINFO_DATA devInfoData;
        devInfoData.cbSize = sizeof(SP_DEVINFO_DATA);
        
        if (SetupDiGetDeviceInterfaceDetailW(hDevInfo, &interfaceData, detailData, 
                                              requiredSize, nullptr, &devInfoData)) {
            // 获取设备实例 ID

            if (CM_Get_Device_IDW(devInfoData.DevInst, instanceId, MAX_PATH, 0) == CR_SUCCESS) {
```

```cpp
// IsoImageManager.cpp (ISO 解析核心)

IsoInfo IsoImageManager::parseIso(const QString& filePath) {
    IsoInfo info;
    info.filePath = filePath;
    info.fileName = QFileInfo(filePath).fileName();
    info.fileSize = QFileInfo(filePath).size();
    
    QFile file(filePath);
    if (!file.open(QIODevice::ReadOnly)) {
        info.errorString = tr("Cannot open file: %1").arg(file.errorString());
        return info;
    }
    
    // 读取 ISO 主卷描述符 (LBA 16, 2048 bytes/sector)
    file.seek(16 * 2048);
    QByteArray sector = file.read(2048);
    
    if (sector.size() < 2048) {
        info.errorString = tr("File too small to be a valid ISO");
        return info;
    }
    
    // 检查 ISO 9660 魔数
    if (sector[0] != 0x01 || sector[1] != 'C' || sector[2] != 'D' || 
        sector[3] != '0' || sector[4] != '0' || sector[5] != '1') {
        info.errorString = tr("Invalid ISO 9660 signature");
        return info;
    }
    
    // 提取卷标 (偏移 40, 32 bytes)
    QByteArray volumeLabel = sector.mid(40, 32);
    // 去除尾部空格
    int endPos = volumeLabel.indexOf('\x00');
    if (endPos < 0) endPos = volumeLabel.size();
    while (endPos > 0 && volumeLabel[endPos - 1] == ' ') endPos--;
    info.displayName = QString::fromLatin1(volumeLabel.left(endPos));
    
    // 检测启动模式
    info.bootMode = detectBootMode(filePath);
    
    // 检测架构 (通过检查 ISO 中的文件)
    info.arch = detectArch(filePath);
    
    // 尝试提取 Windows 版本信息
    info.version = extractVersionFromIso(filePath);
    
    info.isValid = true;
    return info;
}

BootMode IsoImageManager::detectBootMode(const QString& filePath) {
    QFile file(filePath);
    if (!file.open(QIODevice::ReadOnly)) return BootMode::Unknown;
    
    // 检查 El Torito 启动目录
    // El Torito 引导记录在 LBA 17
    file.seek(17 * 2048);
    QByteArray bootRecord = file.read(2048);
    
    if (bootRecord.size() >= 8 && 
        bootRecord.mid(0, 8) == "EL TORITO SPECIFICATION") {
        
        // 检查 UEFI 启动文件 (efi/boot/bootx64.efi 或 efi/boot/bootia32.efi)
        // 简化的检查: 搜索常见 UEFI 文件路径
        file.seek(0);
        QByteArray header = file.read(65536);  // 读取前 64KB 搜索路径
        
        bool hasUefi = header.contains("efi\\boot\\bootx64.efi") ||
                        header.contains("EFI/BOOT/BOOTX64.EFI") ||
                        header.contains("efi\\boot\\bootia32.efi");
        
        bool hasBios = header.contains("isolinux.bin") ||
                       header.contains("bootmgr") ||
                       header.contains("ntldr");
        
        if (hasUefi && hasBios) return BootMode::Dual;
        if (hasUefi) return BootMode::UEFI;
        if (hasBios) return BootMode::BIOS;
    }
    
    return BootMode::Unknown;
}

ArchType IsoImageManager::detectArch(const QString& filePath) {
    QFile file(filePath);
    if (!file.open(QIODevice::ReadOnly)) return ArchType::Unknown;
    
    // 搜索架构标识
    QByteArray header = file.read(262144);  // 256KB
    
    if (header.contains("amd64") || header.contains("x64") || 
        header.contains("BOOTX64.EFI")) {
        return ArchType::x64;
    }
    if (header.contains("x86") || header.contains("i386") || 
        header.contains("BOOTIA32.EFI")) {
        return ArchType::x86;
    }
    if (header.contains("arm64") || header.contains("aarch64") ||
        header.contains("BOOTAA64.EFI")) {
        return ArchType::ARM64;
    }
    
    return ArchType::Unknown;
}
```

```cpp
// FormatManager.cpp (核心写入逻辑)

void FormatManager::startCreation(const UsbDevice& device, const IsoInfo& iso,
                                   const FormatOptions& options) {
    if (m_running) return;
    
    m_cancelled = false;
    m_running = true;
    m_bytesWritten = 0;
    
    // 在工作线程中执行
    m_workerThread = QThread::create([this, device, iso, options]() {
        runCreation(device, iso, options);
    });
    
    connect(m_workerThread, &QThread::finished, [this]() {
        m_running = false;
        m_workerThread->deleteLater();
        m_workerThread = nullptr;
    });
    
    m_workerThread->start();
}

void FormatManager::runCreation(const UsbDevice& device, const IsoInfo& iso,
                                 const FormatOptions& options) {
    m_timer.start();
    
    emit stageChanged(FormatStage::Preparing, tr("Preparing device..."));
    emit logMessage(tr("Starting creation process for %1").arg(device.driveLetter));
    
    // 1. 锁定并卸载卷
    if (!lockAndDismountVolume(device.driveLetter)) {
        emit errorOccurred(tr("Failed to lock volume %1. Is it in use?").arg(device.driveLetter));
        emit finished(false, tr("Volume lock failed"));
        return;
    }
    
    // 2. 格式化设备
    emit stageChanged(FormatStage::WritingMBR, tr("Formatting device..."));
    if (!formatDevice(device, options)) {
        if (!m_cancelled) {
            emit errorOccurred(tr("Format failed"));
            emit finished(false, tr("Format failed"));
        }
        return;
    }
    
    if (m_cancelled) {
        emit stageChanged(FormatStage::Cancelled, tr("Cancelled by user"));
        emit finished(false, tr("Cancelled"));
        return;
    }
    
    // 3. 写入引导扇区 (MBR/GPT)
    emit stageChanged(FormatStage::WritingMBR, tr("Writing boot records..."));
    if (!writeBootSector(device, options)) {
        emit errorOccurred(tr("Boot sector write failed"));
        emit finished(false, tr("Boot sector failed"));
        return;
    }
    
    // 4. 复制 ISO 文件
    emit stageChanged(FormatStage::CopyingFiles, tr("Copying ISO files..."));
    if (!copyIsoFiles(device, iso)) {
        if (!m_cancelled) {
            emit errorOccurred(tr("File copy failed"));
            emit finished(false, tr("File copy failed"));
        }
        return;
    }
    
    // 5. 安装 UEFI:NTFS 驱动 (如果需要)
    if (options.partitionScheme == PartitionScheme::GPT ||
        (iso.bootMode == BootMode::UEFI || iso.bootMode == BootMode::Dual)) {
        emit stageChanged(FormatStage::InstallingDrivers, tr("Installing UEFI drivers..."));
        installUefiDrivers(device);
    }
    
    // 6. 完成
    emit stageChanged(FormatStage::Finalizing, tr("Finalizing..."));
    unlockVolume(device.driveLetter);
    
    int elapsed = m_timer.elapsed() / 1000;
    emit logMessage(tr("Process completed in %1 seconds").arg(elapsed));
    emit finished(true, tr("Bootable USB created successfully!"));
}

bool FormatManager::formatDevice(const UsbDevice& device, const FormatOptions& options) {
    // 使用 Windows FormatEx API 或 VDS (Virtual Disk Service)
    // 这里展示使用 FormatEx (fmifs.dll) 的方式
    
    using PFormatEx = BOOL (WINAPI*)(PWCHAR, DWORD, MEDIA_TYPE, PWCHAR, 
                                      BOOL, DWORD, PFMIFSCALLBACK);
    
    HMODULE hFmifs = LoadLibraryW(L"fmifs.dll");
    if (!hFmifs) return false;
    
    PFormatEx formatEx = reinterpret_cast<PFormatEx>(
        GetProcAddress(hFmifs, "FormatEx")
    );
    if (!formatEx) {
        FreeLibrary(hFmifs);
        return false;
    }
    
    // 回调函数用于进度报告
    auto callback = [](CALLBACKCOMMAND command, DWORD subCommand, 
                      PVOID commandInfo) -> BOOLEAN {
        // 处理格式化进度回调
        return TRUE;  // 继续
    };
    
    WCHAR driveRoot[4] = {0};
    device.driveLetter.toWCharArray(driveRoot);
    driveRoot[2] = L'\\';
    driveRoot[3] = L'\0';
    
    WCHAR label[32] = {0};
    options.volumeLabel.left(31).toWCharArray(label);
    
    // 文件系统类型映射
    DWORD fsType = 0;
    switch (options.fileSystem) {
        case FileSystem::FAT32: fsType = FMIFS_FAT32; break;
        case FileSystem::NTFS:  fsType = FMIFS_NTFS; break;
        case FileSystem::exFAT: fsType = FMIFS_EXFAT; break;
        default: fsType = FMIFS_FAT32;
    }
    
    BOOL result = formatEx(driveRoot, fsType, RemovableMedia, label,
                           options.quickFormat ? TRUE : FALSE, 0,
                           callback);
    
    FreeLibrary(hFmifs);
    return result == TRUE;
}

bool FormatManager::copyIsoFiles(const UsbDevice& device, const IsoInfo& iso) {
    // 挂载 ISO 并复制文件
    // 方法 1: 使用 Windows 虚拟磁盘 API ( virtdisk.h ) 挂载 ISO
    // 方法 2: 使用 7z/diskimage 库直接读取 ISO 内容
    
    // 这里展示使用 Windows 虚拟磁盘挂载的方式
    QString isoPath = iso.filePath;
    QString mountPoint;
    
    // 打开虚拟磁盘
    VIRTUAL_STORAGE_TYPE storageType;
    storageType.DeviceId = VIRTUAL_STORAGE_TYPE_DEVICE_ISO;
    storageType.VendorId = VIRTUAL_STORAGE_TYPE_VENDOR_MICROSOFT;
    
    HANDLE vhdHandle;
    WCHAR isoPathW[MAX_PATH];
    isoPath.toWCharArray(isoPathW);
    
    OPEN_VIRTUAL_DISK_PARAMETERS openParams;
    openParams.Version = OPEN_VIRTUAL_DISK_VERSION_1;
    
    DWORD ret = OpenVirtualDisk(&storageType, isoPathW, 
                                 VIRTUAL_DISK_ACCESS_READ, 
                                 OPEN_VIRTUAL_DISK_FLAG_NONE,
                                 &openParams, &vhdHandle);
    
    if (ret != ERROR_SUCCESS) {
        emit logMessage(tr("Failed to mount ISO, error: %1").arg(ret));
        // 回退到直接文件复制 (如果 ISO 已挂载)
        return copyFilesFromMountedIso(device, iso);
    }
    
    // 附加虚拟磁盘
    ATTACH_VIRTUAL_DISK_PARAMETERS attachParams;
    attachParams.Version = ATTACH_VIRTUAL_DISK_VERSION_1;
    
    ret = AttachVirtualDisk(vhdHandle, nullptr, 
                            ATTACH_VIRTUAL_DISK_FLAG_READ_ONLY,
                            0, &attachParams, nullptr);
    
    if (ret != ERROR_SUCCESS) {
        CloseHandle(vhdHandle);
        return false;
    }
    
    // 获取挂载点
    WCHAR mountPaths[1024];
    DWORD mountPathSize = 1024;
    GetVirtualDiskPhysicalPath(vhdHandle, &mountPathSize, mountPaths);
    
    // 查找实际盘符 (通过查询设备接口)
    // ... 省略盘符查找逻辑 ...
    
    // 复制文件
    QString sourcePath = mountPoint + "\\*";
    QString destPath = device.driveLetter + "\\";
    
    bool copySuccess = copyDirectoryRecursively(sourcePath, destPath);
    
    // 卸载 ISO
    DetachVirtualDisk(vhdHandle, DETACH_VIRTUAL_DISK_FLAG_NONE, 0);
    CloseHandle(vhdHandle);

}

bool FormatManager::copyDirectoryRecursively(const QString& src, const QString& dst) {
    QDir sourceDir(src);
    if (!sourceDir.exists()) return false;
    
    QDir destDir(dst);
    if (!destDir.exists()) {
        destDir.mkpath(".");
    }
    
    for (const QString& file : sourceDir.entryList(QDir::Files)) {
        if (m_cancelled) return false;
        
        QString srcPath = sourceDir.absoluteFilePath(file);
        QString dstPath = destDir.absoluteFilePath(file);
        
        emit logMessage(tr("Copying: %1").arg(file));
        
        if (!QFile::copy(srcPath, dstPath)) {
            // 如果失败，尝试使用 Windows CopyFileEx 以支持进度回调
            if (!copyFileWithProgress(srcPath, dstPath)) {
                emit errorOccurred(tr("Failed to copy: %1").arg(file));
                return false;
            }
        }
        
        m_bytesWritten += QFileInfo(srcPath).size();
        updateProgress();
    }
    
    for (const QString& dir : sourceDir.entryList(QDir::Dirs | QDir::NoDotAndDotDot)) {
        if (m_cancelled) return false;
        
        QString srcPath = sourceDir.absoluteFilePath(dir);
        QString dstPath = destDir.absoluteFilePath(dir);
        
        if (!copyDirectoryRecursively(srcPath, dstPath)) {
            return false;
        }
    }
    
    return true;
}

bool FormatManager::copyFileWithProgress(const QString& src, const QString& dst) {
    using PCopyFileEx = BOOL (WINAPI*)(LPCWSTR, LPCWSTR, 
                                        LPPROGRESS_ROUTINE, LPVOID, 
                                        LPBOOL, DWORD);
    
    HMODULE hKernel = GetModuleHandleW(L"kernel32.dll");
    PCopyFileEx copyFileEx = reinterpret_cast<PCopyFileEx>(
        GetProcAddress(hKernel, "CopyFileExW")
    );
    if (!copyFileEx) return false;
    
    // 进度回调
    auto progressCallback = [](LARGE_INTEGER totalSize, LARGE_INTEGER transferred,
                              LARGE_INTEGER streamSize, LARGE_INTEGER streamTransferred,
                              DWORD dwStreamNumber, DWORD dwCallbackReason,
                              HANDLE hSourceFile, HANDLE hDestinationFile,
                              LPVOID lpData) -> DWORD {
        if (lpData) {
            auto* mgr = static_cast<FormatManager*>(lpData);
            int percent = static_cast<int>(transferred.QuadPart * 100 / totalSize.QuadPart);
            mgr->emitProgress(percent);
        }
        return PROGRESS_CONTINUE;
    };
    
    BOOL cancelled = FALSE;
    BOOL result = copyFileEx(src.toStdWString().c_str(),
                              dst.toStdWString().c_str(),
                              progressCallback, this,
                              &cancelled, COPY_FILE_NO_BUFFERING);
    
    return result == TRUE;
}
```

```cpp
HANDLE Win32Disk::openPhysicalDisk(int diskNumber, bool writeAccess) {
    QString path = QString("\\\\.\\PhysicalDrive%1").arg(diskNumber);
    HANDLE hDevice = CreateFileW(
        path.toStdWString().c_str(),
        writeAccess ? GENERIC_READ | GENERIC_WRITE : GENERIC_READ,
        FILE_SHARE_READ | FILE_SHARE_WRITE,
        nullptr,
        OPEN_EXISTING,
        FILE_FLAG_NO_BUFFERING | FILE_FLAG_WRITE_THROUGH,
        nullptr
    );
    return hDevice;
}

bool Win32Disk::lockVolume(const QString& driveLetter) {
    QString volumePath = QString("\\\\.\\%1:").arg(driveLetter.left(1));
    HANDLE hVolume = CreateFileW(
        volumePath.toStdWString().c_str(),
        GENERIC_READ | GENERIC_WRITE,
        FILE_SHARE_READ | FILE_SHARE_WRITE,
        nullptr, OPEN_EXISTING, 0, nullptr
    );
    
    if (hVolume == INVALID_HANDLE_VALUE) return false;
    
    DWORD bytesReturned;
    BOOL result = DeviceIoControl(hVolume, FSCTL_LOCK_VOLUME,
                                   nullptr, 0, nullptr, 0,
                                   &bytesReturned, nullptr);
    if (result) {
        // 卸载卷
        DeviceIoControl(hVolume, FSCTL_DISMOUNT_VOLUME,
                        nullptr, 0, nullptr, 0,
                        &bytesReturned, nullptr);
    }
    
    CloseHandle(hVolume);
    return result == TRUE;
}

bool Win32Disk::writeMBR(int diskNumber, const QByteArray& mbrData) {
    HANDLE hDisk = openPhysicalDisk(diskNumber, true);
    if (hDisk == INVALID_HANDLE_VALUE) return false;
    
    // MBR 是磁盘前 512 字节
    DWORD bytesWritten;
    BOOL result = WriteFile(hDisk, mbrData.constData(), 
                            qMin<qint64>(mbrData.size(), 512),
                            &bytesWritten, nullptr);
    
    if (result) FlushFileBuffers(hDisk);
    CloseHandle(hDisk);
    return result == TRUE && bytesWritten == 512;
}
```

```cpp
// HashUtils.cpp
#include <wincrypt.h>

QByteArray HashUtils::calculateSha256(const QString& filePath,
                                       std::function<bool()> isCancelled) {
    QFile file(filePath);
    if (!file.open(QIODevice::ReadOnly)) return QByteArray();
    
    HCRYPTPROV hProv = 0;
    HCRYPTHASH hHash = 0;
    
    if (!CryptAcquireContextW(&hProv, nullptr, nullptr, 
                              PROV_RSA_AES, CRYPT_VERIFYCONTEXT)) {
        return QByteArray();
    }
    
    if (!CryptCreateHash(hProv, CALG_SHA_256, 0, 0, &hHash)) {
        CryptReleaseContext(hProv, 0);
        return QByteArray();
    }
    
    const qint64 bufferSize = 1024 * 1024;  // 1MB 缓冲区
    QByteArray buffer;
    qint64 totalRead = 0;
    qint64 fileSize = file.size();
    int lastPercent = -1;
    
    while (!file.atEnd()) {
        if (isCancelled && isCancelled()) {
            CryptDestroyHash(hHash);
            CryptReleaseContext(hProv, 0);
            return QByteArray();
        }
        
        buffer = file.read(bufferSize);
        if (!CryptHashData(hHash, 
                           reinterpret_cast<const BYTE*>(buffer.constData()),
                           buffer.size(), 0)) {
            break;
        }
        
        totalRead += buffer.size();
        int percent = static_cast<int>(totalRead * 100 / fileSize);
        if (percent != lastPercent) {
            lastPercent = percent;
            // 可以在这里发送进度信号
        }
    }
    
    DWORD hashLen = 0;
    DWORD hashLenSize = sizeof(DWORD);
    CryptGetHashParam(hHash, HP_HASHSIZE, 
                      reinterpret_cast<BYTE*>(&hashLen), &hashLenSize, 0);
    
    QByteArray hash(hashLen, Qt::Uninitialized);
    CryptGetHashParam(hHash, HP_HASHVAL, 
                      reinterpret_cast<BYTE*>(hash.data()), &hashLen, 0);
    
    CryptDestroyHash(hHash);
    CryptReleaseContext(hProv, 0);
    
    return hash.toHex();
}
```

```batch
@echo off

set BUILD_DIR=build\Release

rem 1. 构建 Release 版本
cmake --build build --config Release

rem 2. 部署 Qt 依赖
%QT_DIR%\bin\windeployqt.exe %BUILD_DIR%\FlashBoot.exe

rem 3. 使用 Enigma Virtual Box 打包 (命令行)
"C:\Program Files\Enigma Virtual Box\enigmavbconsole.exe" ^
    -input "%BUILD_DIR%\FlashBoot.exe" ^
    -output "dist\FlashBoot_Portable.exe" ^
    -map "%BUILD_DIR%\*.dll" ^
    -map "%BUILD_DIR%\platforms" ^
    -map "%BUILD_DIR%\styles" ^
    -map "%BUILD_DIR%\translations"
```
